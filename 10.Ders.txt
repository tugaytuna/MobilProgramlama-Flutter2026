  print(sayilar.contains(20));

  // listeyi sıralama - küçükten büyüğe doğru

  List<int> numbers = [0];

  for (int i = 1; i <= 10; i++) {
    numbers.add(i);
  }

  print(numbers);

  for (int i = 0; i < numbers.length; i++) {
    numbers[i] = numbers[i] * numbers[i];
  }

  print(numbers);

  List<int> notlar = [67, 83, 78, 45, 56];

  int toplam = 0;

  // foreach
  // dizinin içindeki eleman tipi
  // istediğimiz isim (değişken oluşturmak gibi)
  // in
  // dizinin adı
  for (int not in notlar) {
    toplam += not;
  }

  print(toplam / notlar.length);
